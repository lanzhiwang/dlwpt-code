```python
import os
import tempfile
import torch
import torch.distributed as dist
import torch.nn as nn
import torch.optim as optim
import torch.multiprocessing as mp

from torch.nn.parallel import DistributedDataParallel as DDP


# -------------------------------------------------------------------
# 1. 初始化与销毁函数: 适配多硬件平台的分布式环境配置
#
# On Windows platform, the torch.distributed package only
# supports Gloo backend, FileStore and TcpStore.
# For FileStore, set init_method parameter in init_process_group
# to a local file. Example as follow:
# init_method="file:///f:/libtmp/some_file"
# dist.init_process_group(
#    "gloo",
#    rank=rank,
#    init_method=init_method,
#    world_size=world_size)
# For TcpStore, same way as on Linux.
#
# -------------------------------------------------------------------
def setup(rank, world_size):
    # 配置主节点的 IP 和端口, 用于进程间的 TCP 握手寻址
    os.environ["MASTER_ADDR"] = "localhost"
    os.environ["MASTER_PORT"] = "12355"

    # We want to be able to train our model on an `accelerator <https://pytorch.org/docs/stable/torch.html#accelerators>`__
    # such as CUDA, MPS, MTIA, or XPU.
    # [深度解析]: PyTorch 近期引入了 torch.accelerator 通用硬件抽象接口.
    # 无论底层是 CUDA(NVIDIA GPU)、XPU(Intel)、MTIA(Meta) 还是 MPS(Apple),
    # current_accelerator() 都能自动检测当前硬件类型.
    acc = torch.accelerator.current_accelerator()

    # 自动获取该硬件对应的默认分布式通信后端(例如 CUDA 对应 "nccl", CPU/Gloo 对应 "gloo")
    # 为什么这么写: 增强代码的跨平台移植性, 避免硬编码 "nccl".
    backend = torch.distributed.get_default_backend_for_device(acc)

    # initialize the process group
    # 正式初始化分布式进程组
    dist.init_process_group(backend, rank=rank, world_size=world_size)


def cleanup():
    # 销毁进程组, 释放底层网络 Socket 连接与显存资源
    dist.destroy_process_group()


# -------------------------------------------------------------------
# 2. 模型定义: 简单的两层 MLP 模型
# -------------------------------------------------------------------
class ToyModel(nn.Module):
    def __init__(self):
        super(ToyModel, self).__init__()
        self.net1 = nn.Linear(10, 10)
        self.relu = nn.ReLU()
        self.net2 = nn.Linear(10, 5)

    def forward(self, x):
        return self.net2(self.relu(self.net1(x)))


# -------------------------------------------------------------------
# 3. 示例 1: 基础 DDP 训练流程 (demo_basic)
# -------------------------------------------------------------------
def demo_basic(rank, world_size):
    print(f"Running basic DDP example on rank {rank}.")
    setup(rank, world_size)

    # create model and move it to GPU with id rank
    # 将模型搬运到当前 Rank 对应的硬件设备上 (例如 cuda:0, cuda:1)
    model = ToyModel().to(rank)
    # 将模型包装为 DDP 模型.
    # device_ids=[rank]: 告诉 DDP 当前进程的模型只在编号为 rank 的这单个设备上.
    ddp_model = DDP(model, device_ids=[rank])

    loss_fn = nn.MSELoss()
    optimizer = optim.SGD(ddp_model.parameters(), lr=0.001)

    optimizer.zero_grad()

    # [细节解释]: 注意这里的 torch.randn(20, 10) 并没有显式调用 .to(rank)!
    # 为什么能正常运行? 因为当 DDP 设定了 device_ids=[rank] 时, 如果传入的输入张量在 CPU 上,
    # DDP 的 forward 函数会自动将 CPU 张量复制移动到 target device (即 rank 卡) 上!
    outputs = ddp_model(torch.randn(20, 10))

    # 标签张量显式移动到对应的设备上
    labels = torch.randn(20, 5).to(rank)

    # 执行反向传播: DDP 自动在后台异步触发 AllReduce 操作, 对所有 Rank 的梯度求平均
    loss_fn(outputs, labels).backward()
    optimizer.step()

    cleanup()
    print(f"Finished running basic DDP example on rank {rank}.")


def run_demo(demo_fn, world_size):
    # 使用 mp.spawn 派生子进程, 将 rank (0 ~ world_size-1) 作为第一个参数自动传给 demo_fn
    mp.spawn(demo_fn, args=(world_size,), nprocs=world_size, join=True)


# -------------------------------------------------------------------
# 4. 示例 2: 分布式环境下的 Checkpoint 保存与加载 (demo_checkpoint)
# -------------------------------------------------------------------
def demo_checkpoint(rank, world_size):
    print(f"Running DDP checkpoint example on rank {rank}.")
    setup(rank, world_size)

    model = ToyModel().to(rank)
    ddp_model = DDP(model, device_ids=[rank])

    CHECKPOINT_PATH = tempfile.gettempdir() + "/model.checkpoint"

    # [最佳实践]: 只在 Rank 0 上保存 Checkpoint!
    # 为什么这么写: 因为在 DDP 初始化时, Rank 0 的权重会被广播给所有进程;
    # 且每次反向传播后所有进程的梯度一致, 故所有进程的模型参数时刻保持一致.
    # 只让 Rank 0 写磁盘可以避免多进程同时写同一个文件导致的 I/O 冲突和磁盘写坏.
    if rank == 0:
        # All processes should see same parameters as they all start from same
        # random parameters and gradients are synchronized in backward passes.
        # Therefore, saving it in one process is sufficient.
        torch.save(ddp_model.state_dict(), CHECKPOINT_PATH)

    # Use a barrier() to make sure that process 1 loads the model after process
    # 0 saves it.
    # [显式屏障同步]: dist.barrier()
    # 为什么这么写: 必须阻塞 Rank 1、Rank 2 等其他进程, 直到 Rank 0 彻底完成写磁盘操作!
    # 否则其他进程过早执行后面的 torch.load() 会因为文件没写完而报错 FileNotFound 或损坏.
    dist.barrier()

    # We want to be able to train our model on an `accelerator <https://pytorch.org/docs/stable/torch.html#accelerators>`__
    # such as CUDA, MPS, MTIA, or XPU.
    acc = torch.accelerator.current_accelerator()

    # configure map_location properly
    # [设备映射(map_location)]: 定义重定向规则.
    # 为什么这么写: Rank 0 保存的文件中, 张量记录的设备是 f"{acc}:0" (比如 cuda:0).
    # 如果 Rank 1 直接 load, 默认会把参数加载到 cuda:0 上然后再搬到 cuda:1, 这会导致 cuda:0 显存突增(OOM 隐患).
    # 使用 map_location={cuda:0 -> cuda:rank} 可以直接把权重反序列化到当前进程对应的卡上!
    map_location = {f"{acc}:0": f"{acc}:{rank}"}

    # weights_only=True: PyTorch 安全机制, 防止 pickle 反序列化代码注入攻击
    ddp_model.load_state_dict(
        torch.load(CHECKPOINT_PATH, map_location=map_location, weights_only=True)
    )

    loss_fn = nn.MSELoss()
    optimizer = optim.SGD(ddp_model.parameters(), lr=0.001)

    optimizer.zero_grad()
    outputs = ddp_model(torch.randn(20, 10))
    labels = torch.randn(20, 5).to(rank)

    loss_fn(outputs, labels).backward()
    optimizer.step()

    # Not necessary to use a dist.barrier() to guard the file deletion below
    # as the AllReduce ops in the backward pass of DDP already served as
    # a synchronization.
    # [隐式屏障同步]: 为什么这里删除文件之前不需要写 dist.barrier()?
    # 答: 因为上面的 loss_fn(...).backward() 内部触发了跨卡 AllReduce 梯度同步!
    # AllReduce 操作本身就是一个强同步点(所有进程必须都到达通信点才能完成).
    # 因此能保证所有进程都已经完成了前向和反向计算, 此时 Rank 0 安全删除临时文件不会影响其他卡.
    if rank == 0:
        os.remove(CHECKPOINT_PATH)

    cleanup()
    print(f"Finished running DDP checkpoint example on rank {rank}.")


# -------------------------------------------------------------------
# 5. 示例 3: DDP 与模型并行(流水线/跨卡切分)结合 (demo_model_parallel)
# -------------------------------------------------------------------
# 定义一个单进程内部跨 2 张 GPU 的模型(模型并行 /流水线并行)
class ToyMpModel(nn.Module):
    def __init__(self, dev0, dev1):
        super(ToyMpModel, self).__init__()
        self.dev0 = dev0
        self.dev1 = dev1
        # 网络第一层放在 dev0 (例如 cuda:0)
        self.net1 = torch.nn.Linear(10, 10).to(dev0)
        self.relu = torch.nn.ReLU()
        # 网络第二层放在 dev1 (例如 cuda:1)
        self.net2 = torch.nn.Linear(10, 5).to(dev1)

    def forward(self, x):
        # 数据先推到 dev0 计算前半部分
        x = x.to(self.dev0)
        x = self.relu(self.net1(x))
        # 中间激活值推到 dev1 计算后半部分(跨卡传输)
        x = x.to(self.dev1)
        return self.net2(x)


def demo_model_parallel(rank, world_size):
    print(f"Running DDP with model parallel example on rank {rank}.")
    setup(rank, world_size)

    # setup mp_model and devices for this process
    # 假设每台机器有 4 张 GPU, 每个进程分配 2 张 GPU:
    # Rank 0 使用 dev0=cuda:0, dev1=cuda:1
    # Rank 1 使用 dev0=cuda:2, dev1=cuda:3
    dev0 = rank * 2
    dev1 = rank * 2 + 1
    mp_model = ToyMpModel(dev0, dev1)

    # [关键细节]: 这里的 DDP 包装没有传 device_ids 参数! 即 DDP(mp_model)!
    # 为什么这么写: 因为 mp_model 是一个跨多张 GPU (dev0 和 dev1) 的模型!
    # 如果传了 device_ids=[dev0], DDP 会误以为模型全部在 dev0 上而报错.
    # 当不传 device_ids 时, DDP 会自动识别并支持这种跨设备的模型并行架构,
    # 在反向传播时精确对位于不同 GPU 上的参数各自发起梯度同步.
    ddp_mp_model = DDP(mp_model)

    loss_fn = nn.MSELoss()
    optimizer = optim.SGD(ddp_mp_model.parameters(), lr=0.001)

    optimizer.zero_grad()

    # outputs will be on dev1
    outputs = ddp_mp_model(torch.randn(20, 10))

    # 模型的最终输出在 dev1 上, 因此 Loss 的 Target label 也必须移动到 dev1!
    labels = torch.randn(20, 5).to(dev1)
    loss_fn(outputs, labels).backward()
    optimizer.step()

    cleanup()
    print(f"Finished running DDP with model parallel example on rank {rank}.")


# -------------------------------------------------------------------
# 6. 主程序入口
# -------------------------------------------------------------------
if __name__ == "__main__":
    # 检测系统可用的加速器数量
    n_gpus = torch.accelerator.device_count()
    assert n_gpus >= 2, f"Requires at least 2 GPUs to run, but got {n_gpus}"
    world_size = n_gpus

    # 场景 1 & 场景 2: 标准 DDP, 1 个进程对应 1 张 GPU(例如 4 卡开启 4 个进程)
    run_demo(demo_basic, world_size)
    run_demo(demo_checkpoint, world_size)

    # 场景 3: 混合并行, 1 个进程对应 2 张 GPU(例如 4 卡只需开启 2 个进程)
    world_size = n_gpus // 2
    run_demo(demo_model_parallel, world_size)

```

### 二、上述示例代码所表达的 DDP 核心特性

这份官方示例代码不仅演示了基本的用法, 更深刻地阐释了 PyTorch DDP 在实际大模型训练中的 4 大核心能力与设计哲学:

#### 1. 硬件无关性与统一抽象 (Hardware-Agnostic Abstraction)

通过 `torch.accelerator` 与 `torch.distributed.get_default_backend_for_device()`, 展示了 DDP 架构的通用性. DDP 的分布式数据并行协议(进程组管理、梯度 Bucketing、AllReduce)完全独立于底层硬件类型:

* 在 NVIDIA 上使用 CUDA + NCCL;

* 在 Intel 芯片上使用 XPU + OneCCL;

* 在 Apple 芯片上使用 MPS + Gloo.

上层 PyTorch DDP 代码可以做到一套代码, 跨硬件无缝运行.

#### 2. 分布式文件系统协同与正确的状态恢复机制 (Distributed Checkpointing)

`demo_checkpoint` 揭示了多进程环境中保存与加载模型的两大黄金法则:

* 单一写角色与显式屏障(Explicit Barrier): 只有 `Rank 0` 拥有写文件的权限, 结合 `dist.barrier()` 确保写操作完成后其他卡才去读取, 彻底避免多进程文件读写竞争(Race Condition).

* 精准设备重定向(`map_location` 机制): 反序列化模型权重时, 必须使用字典映射明确把权重加载到当前 Rank 对应的设备上, 防止所有卡同时将权重 Load 到 `cuda:0` 导致的爆显存(OOM).

#### 3. 通信原语的隐式同步效应 (Implicit Synchronization via AllReduce)

`demo_checkpoint` 的结尾处在未加 `dist.barrier()` 的情况下直接由 Rank 0 删除了临时文件. 这展现了 DDP 的一个底层特性:

* `loss.backward()` 内部调用的集合通信原语(AllReduce)在数学和逻辑上构成了一个隐式屏障(Implicit Barrier). 所有进程必须都计算完梯度并完成通信才能退出反向传播, 因此反向传播之后可以安全地假定"所有进程动作已同步".

#### 4. DDP 与模型并行(Model Parallelism)的混合并行能力

`demo_model_parallel` 展示了 DDP 最强大的扩展性之一: 混合并行(Hybrid Parallelism).

* 场景: 当单个模型太大, 1 张 GPU 装不下时, 我们可以将模型切分到 2 张 GPU 上(流水线并行/模型并行), 即 `1 个进程控制 2 张 GPU`.

* DDP 的兼容性: DDP 并不强制要求 `1 个进程只能管 1 张 GPU`. 当不传 `device_ids` 时, DDP 能自动识别单进程内部跨多卡的复杂模型结构, 并在跨进程间精准完成不同设备上梯度的 AllReduce 同步.

* 在 LLM 时代, 这种模式是 DP + PP (Data Parallelism + Pipeline Parallelism) 混合并行架构的基石.

### 脚本 1: 基础 DDP 训练流程 (`ddp_demo1_basic.py`)

```python
import os
import torch
import torch.distributed as dist
import torch.nn as nn
import torch.optim as optim
from torch.nn.parallel import DistributedDataParallel as DDP


# -------------------------------------------------------------------
# 1. 简单模型定义
# -------------------------------------------------------------------
class ToyModel(nn.Module):
    def __init__(self):
        super(ToyModel, self).__init__()
        self.net1 = nn.Linear(10, 10)
        self.relu = nn.ReLU()
        self.net2 = nn.Linear(10, 5)

    def forward(self, x):
        return self.net2(self.relu(self.net1(x)))


# -------------------------------------------------------------------
# 2. 主训练逻辑
# -------------------------------------------------------------------
def main():
    # 从 torchrun 环境变量中读取进程和设备信息
    # local_rank: 当前节点(单机)内的 GPU 编号 (0~7)
    # global_rank: 整个 2 机 16 卡集群中的唯一进程 ID (0~15)
    # world_size: 全局总卡数 (16)
    local_rank = int(os.environ["LOCAL_RANK"])
    global_rank = int(os.environ["RANK"])
    world_size = int(os.environ["WORLD_SIZE"])

    print(f"[Global Rank {global_rank} | Local Rank {local_rank}] 启动基础 DDP 示例...")

    # 绑定本地设备并初始化分布式进程组
    torch.cuda.set_device(local_rank)
    dist.init_process_group(backend="nccl")

    # 创建模型并复制到本地 GPU
    model = ToyModel().to(local_rank)
    # 包装为 DDP 模型, device_ids 绑定当前进程的 GPU
    ddp_model = DDP(model, device_ids=[local_rank])

    loss_fn = nn.MSELoss()
    optimizer = optim.SGD(ddp_model.parameters(), lr=0.001)

    optimizer.zero_grad()

    # 模拟数据输入: CPU 张量传入 ddp_model, DDP 会自动将其移动到 local_rank 对应的设备上
    outputs = ddp_model(torch.randn(20, 10))
    labels = torch.randn(20, 5).to(local_rank)

    # 计算损失并反向传播 (触发跨 2 机 16 卡的 AllReduce 梯度求平均同步)
    loss = loss_fn(outputs, labels)
    loss.backward()

    # 更新权重
    optimizer.step()

    # 栅栏同步与资源清理
    dist.barrier()
    dist.destroy_process_group()
    print(f"[Global Rank {global_rank}] 基础训练完成, 通信资源已释放.")


if __name__ == "__main__":
    main()

```

### 脚本 2: 分布式环境下的 Checkpoint 保存与加载 (`ddp_demo2_checkpoint.py`)

> 跨机器重要提示: 在多机多卡(Multi-Node)环境下, 为了让 Node 1 的卡能读取到 Node 0 保存的 Checkpoint 文件, 保存路径推荐使用网络共享存储路径(如 NFS / Ceph 挂载目录). 这里代码使用了共享路径配置.
>

```python
import os
import tempfile
import torch
import torch.distributed as dist
import torch.nn as nn
import torch.optim as optim
from torch.nn.parallel import DistributedDataParallel as DDP


class ToyModel(nn.Module):
    def __init__(self):
        super(ToyModel, self).__init__()
        self.net1 = nn.Linear(10, 10)
        self.relu = nn.ReLU()
        self.net2 = nn.Linear(10, 5)

    def forward(self, x):
        return self.net2(self.relu(self.net1(x)))


def main():
    local_rank = int(os.environ["LOCAL_RANK"])
    global_rank = int(os.environ["RANK"])
    world_size = int(os.environ["WORLD_SIZE"])

    print(
        f"[Global Rank {global_rank} | Local Rank {local_rank}] 启动 Checkpoint 保存与加载示例..."
    )

    torch.cuda.set_device(local_rank)
    dist.init_process_group(backend="nccl")

    model = ToyModel().to(local_rank)
    ddp_model = DDP(model, device_ids=[local_rank])

    # 设置 Checkpoint 文件路径 (多机训练建议配置在共享挂载目录如 /mnt/shared/checkpoint.pth)
    CHECKPOINT_PATH = os.path.join(tempfile.gettempdir(), "ddp_demo_model.checkpoint")

    # ---------------------------------------------------------------
    # 步骤 A: 仅在 Global Rank 0 上保存模型(避免 16 个进程并发写文件冲突)
    # ---------------------------------------------------------------
    if global_rank == 0:
        torch.save(ddp_model.state_dict(), CHECKPOINT_PATH)
        print(f"[Global Rank 0] 模型检查点已保存至: {CHECKPOINT_PATH}")

    # ---------------------------------------------------------------
    # 步骤 B: 显式屏障同步(dist.barrier)
    # 强制阻塞所有其他 15 个进程, 直到 Global Rank 0 彻底写完文件!
    # ---------------------------------------------------------------
    dist.barrier()

    # ---------------------------------------------------------------
    # 步骤 C: 所有进程安全地加载模型
    # ---------------------------------------------------------------
    # map_location 重定向规则: 将 Rank 0 保存时对应的 cuda:0, 重定向反序列化到本进程绑定的 local_rank 卡上
    # 这避免了所有卡加载权重时瞬间挤爆 cuda:0 显存的问题
    map_location = {"cuda:0": f"cuda:{local_rank}"}

    # weights_only=True 为 PyTorch 安全机制, 防止 Pickle 代码注入
    ddp_model.load_state_dict(
        torch.load(CHECKPOINT_PATH, map_location=map_location, weights_only=True)
    )
    print(f"[Global Rank {global_rank}] 权重加载成功!")

    # 执行一次正常的训练 Step
    loss_fn = nn.MSELoss()
    optimizer = optim.SGD(ddp_model.parameters(), lr=0.001)

    optimizer.zero_grad()
    outputs = ddp_model(torch.randn(20, 10))
    labels = torch.randn(20, 5).to(local_rank)

    # 反向传播: 内部自带跨卡 AllReduce 同步机制, 构成了隐式的同步屏障
    loss = loss_fn(outputs, labels)
    loss.backward()
    optimizer.step()

    # ---------------------------------------------------------------
    # 步骤 D: 删除临时文件
    # 借助上面 loss.backward() 完成的隐式同步, 此时可安全由 Global Rank 0 删除文件
    # ---------------------------------------------------------------
    if global_rank == 0:
        if os.path.exists(CHECKPOINT_PATH):
            os.remove(CHECKPOINT_PATH)
            print(f"[Global Rank 0] 临时 Checkpoint 文件已清理.")

    dist.barrier()
    dist.destroy_process_group()


if __name__ == "__main__":
    main()

```

### 脚本 3: DDP 与模型并行(流水线/跨卡切分)结合 (`ddp_demo3_model_parallel.py`)

> 核心逻辑: 单进程控制多张 GPU(如流水线并行), 同时进程间进行 DDP 数据并行.
> 假设每台机器 8 张 GPU, 单个进程需要占用 2 张 GPU, 因此单台机器只需开启 4 个进程(`nproc_per_node=4`), 2 台机器共 8 个进程覆盖 16 张卡.
>

```python
import os
import torch
import torch.distributed as dist
import torch.nn as nn
import torch.optim as optim
from torch.nn.parallel import DistributedDataParallel as DDP


# -------------------------------------------------------------------
# 1. 跨多张 GPU 切分的模型(模型并行 / 流水线并行示例)
# -------------------------------------------------------------------
class ToyMpModel(nn.Module):
    def __init__(self, dev0, dev1):
        super(ToyMpModel, self).__init__()
        self.dev0 = dev0
        self.dev1 = dev1
        # 网络前半部分部署在 dev0 上
        self.net1 = torch.nn.Linear(10, 10).to(dev0)
        self.relu = torch.nn.ReLU()
        # 网络后半部分部署在 dev1 上
        self.net2 = torch.nn.Linear(10, 5).to(dev1)

    def forward(self, x):
        # 数据送入 dev0 计算
        x = x.to(self.dev0)
        x = self.relu(self.net1(x))
        # 中间激活值跨卡传输到 dev1 计算
        x = x.to(self.dev1)
        return self.net2(x)


# -------------------------------------------------------------------
# 2. 主训练逻辑
# -------------------------------------------------------------------
def main():
    # 当 torchrun 以 --nproc_per_node=4 启动时:
    # 本地的 local_rank 范围是 0, 1, 2, 3
    local_rank = int(os.environ["LOCAL_RANK"])
    global_rank = int(os.environ["RANK"])
    world_size = int(os.environ["WORLD_SIZE"])  # 全局总进程数为 8

    # 计算当前进程占用的 2 张本地 GPU 编号:
    # local_rank=0 -> dev0=cuda:0, dev1=cuda:1
    # local_rank=1 -> dev0=cuda:2, dev1=cuda:3
    # local_rank=2 -> dev0=cuda:4, dev1=cuda:5
    # local_rank=3 -> dev0=cuda:6, dev1=cuda:7
    dev0 = local_rank * 2
    dev1 = local_rank * 2 + 1

    # 绑定进程主 GPU 为 dev0
    torch.cuda.set_device(dev0)
    dist.init_process_group(backend="nccl")

    print(f"[Global Rank {global_rank}] 控制本地 GPU: ({dev0}, {dev1})")

    # 构建跨 2 张 GPU 的单进程模型
    mp_model = ToyMpModel(dev0, dev1)

    # [核心要点]: 包装 DDP 时, 不传入 device_ids 参数! 即 DDP(mp_model)
    # 原因: 模型已经横跨了 dev0 和 dev1 两张卡, 传入单个 device_ids 会引发误判.
    # 不传 device_ids 时, DDP 会自动处理位于不同 GPU 上的参数梯度的正确跨进程同步.
    ddp_mp_model = DDP(mp_model)

    loss_fn = nn.MSELoss()
    optimizer = optim.SGD(ddp_mp_model.parameters(), lr=0.001)

    optimizer.zero_grad()

    outputs = ddp_mp_model(torch.randn(20, 10))
    # 模型的最后输出位于 dev1 上, 因此 Labels 也必须放在 dev1 上进行 Loss 计算
    labels = torch.randn(20, 5).to(dev1)

    loss = loss_fn(outputs, labels)
    loss.backward()
    optimizer.step()

    dist.barrier()
    dist.destroy_process_group()
    print(f"[Global Rank {global_rank}] 模型并行 DDP 训练完成.")


if __name__ == "__main__":
    main()

```

模型并行和流水线并行的含义分别是什么? 请用具体的代码示例进行对比说明

在回答这个问题之前, 我们先给出一个明确的结论:

上述代码中的 `ToyMpModel` 在分类上属于广义的"模型并行(Model Parallelism)", 更具体、精确的分类是"流水线并行(Pipeline Parallelism, 简称 PP)".

在早期的 PyTorch 文档和论文中, 官方习惯将所有"把模型参数拆分到多张 GPU 上"的做法统称为 Model Parallelism. 但随着大模型(LLM)的爆发, 业界对模型拆分方式进行了更精准的二次细分: 分为 `流水线并行(Pipeline Parallelism, PP)` 和 `张量并行(Tensor Parallelism, TP)`.

下面为你详细拆解两者的含义、核心区别以及具体的代码对比.

---

### 一、模型并行 vs 流水线并行 的概念区别

#### 1. 广义的模型并行(Model Parallelism, MP)

* 定义: 只要模型的参数被拆分到了不同的 GPU 上(而不是像 DDP 那样每张 GPU 复制一份完整模型), 这种架构就统称为模型并行.

* 主要细分:

  * 层间拆分(Inter-layer Parallelism) $\rightarrow$ 即 `流水线并行(Pipeline Parallelism, PP)`

  * 层内拆分(Intra-layer Parallelism) $\rightarrow$ 即 `张量并行(Tensor Parallelism, TP)`

#### 2. 流水线并行(Pipeline Parallelism, PP) -- 层与层之间切分

* 含义: 按照模型的网络层(Layers)进行切分. 把前几层放到 GPU 0, 中间几层放到 GPU 1, 后几层放到 GPU 2.

* 数据流向: 像工厂的流水线一样, 数据顺序经过 GPU 0 $\rightarrow$ GPU 1 $\rightarrow$ GPU 2.

* 代码特征: 如你引用的代码, `net1` 部署在 `dev0`, `net2` 部署在 `dev1`, 前向传播时, 激活值(Activations)由 `dev0` 顺序传给 `dev1`.

#### 3. 张量并行(Tensor Parallelism, TP) -- 单层内部切分(狭义的模型并行)

* 含义: 单个网络层(如一个巨型的 `nn.Linear` 权重矩阵)本身太大, 1 张 GPU 装不下, 需要把这个矩阵切开(按行切或按列切), 让 GPU 0 存矩阵左半部分, GPU 1 存矩阵右半部分.

* 数据流向: GPU 0 和 GPU 1 同时(并行)计算同一个矩阵乘法的一部分, 最后通过通信(All-Reduce / All-Gather)将结果拼合或相加.

* 典型代表: Megatron-LM 中的 Column Parallel Linear 和 Row Parallel Linear.

---

### 二、核心差异对比表

| 维度     | 流水线并行(Pipeline Parallelism, PP) | 张量并行(Tensor Parallelism, TP)              |
| :------- | :----------------------------------- | :-------------------------------------------- |
| 切分粒度 | 按层切分(Layer-wise)                 | 按算子/矩阵切分(Intra-Operator/Tensor)        |
| 计算模式 | 顺序执行(GPU 0 算完才传给 GPU 1)     | 并行计算(GPU 0 和 GPU 1 同时算各自的矩阵块)   |
| 通信内容 | 传递中间激活值(Activations)          | 传递矩阵运算结果(All-Reduce / All-Gather)     |
| 通信频率 | 较低(每个 Stage 只在边界通信一次)    | 极高(每一个 Transformer Layer 内部通信两次)   |
| 硬件要求 | PCIe / 局域网即可                    | 极高, 强烈依赖 NVLink / NVSwitch 局域高速互连 |

---

### 三、具体的 Python 代码示例对比

为了让你直观理解两者的代码形态差异, 我们写两个最简可运行对比示例(假设单进程控制 `dev0` 和 `dev1` 2 张 GPU):

#### 示例 1: 流水线并行(Pipeline Parallelism, PP)

把一个 2 层的网络按层拆分到 2 张 GPU, 顺序计算.

```python
import torch
import torch.nn as nn


# [流水线并行 PP]: 按层切分 (Layer-wise)
class PipelineParallelModel(nn.Module):

    def __init__(self, dev0, dev1):
        super().__init__()
        self.dev0 = dev0
        self.dev1 = dev1

        # 第一层部署在 GPU 0 (dev0)
        self.layer1 = nn.Linear(100, 100).to(dev0)
        # 第二层部署在 GPU 1 (dev1)
        self.layer2 = nn.Linear(100, 50).to(dev1)

    def forward(self, x):
        # 1. 在 dev0 上计算第 1 层
        x = x.to(self.dev0)
        x = torch.relu(self.layer1(x))

        # 2. 将激活值跨卡传送给 dev1(流水线传递)
        x = x.to(self.dev1)

        # 3. 在 dev1 上计算第 2 层
        x = self.layer2(x)
        return x


# 测试运行
dev0 = "cuda:0" if torch.cuda.is_available() else "cpu"
dev1 = "cuda:1" if torch.cuda.device_count() > 1 else dev0

pp_model = PipelineParallelModel(dev0, dev1)
inputs = torch.randn(16, 100)
outputs = pp_model(inputs)
print("PP 输出 Shape:", outputs.shape)  # [16, 50], 位于 dev1 上

```

#### 示例 2: 张量并行(Tensor Parallelism, TP)

把一个大型 `Linear(100, 200)` 层按列切成两半, 每张 GPU 存一半权重 `Linear(100, 100)`, 并行计算后拼接.

```python
import torch
import torch.nn as nn


# [张量并行 TP]: 单层矩阵切分 (Column Parallel Linear)
class ColumnParallelLinear(nn.Module):

    def __init__(self, in_features, out_features, dev0, dev1):
        super().__init__()
        self.dev0 = dev0
        self.dev1 = dev1

        assert out_features % 2 == 0, "输出维度必须能被 GPU 数量整除"
        half_out = out_features // 2  # 200 切分为两个 100

        # GPU 0 存储权重的左半部分 [100, 100]
        self.part0 = nn.Linear(in_features, half_out).to(dev0)
        # GPU 1 存储权重的右半部分 [100, 100]
        self.part1 = nn.Linear(in_features, half_out).to(dev1)

    def forward(self, x):
        # 1. 将相同的输入广播推送给两张 GPU
        x0 = x.to(self.dev0)
        x1 = x.to(self.dev1)

        # 2. 两位 GPU 同时(并行)计算矩阵乘法的一部分
        out0 = self.part0(x0)  # Shape: [16, 100] located on dev0
        out1 = self.part1(x1)  # Shape: [16, 100] located on dev1

        # 3. 将 dev1 的计算结果传输回 dev0, 沿着最后一个维度拼接汇总
        out1_on_dev0 = out1.to(self.dev0)
        full_out = torch.cat([out0, out1_on_dev0], dim=-1)  # 拼接为 [16, 200]

        return full_out


# 测试运行
tp_model = ColumnParallelLinear(in_features=100, out_features=200, dev0=dev0, dev1=dev1)
inputs = torch.randn(16, 100)
outputs = tp_model(inputs)
print("TP 输出 Shape:", outputs.shape)  # [16, 200], 位于 dev0 上

```

### 四、总结与现代 LLM 的实践

1. 回到原问题: 示例中的 `ToyMpModel` 将 `net1` 移动到 `dev0`, 将 `net2` 移动到 `dev1`, 并在 `forward` 中将激活值从 `dev0` 移动到 `dev1`, 这属于标准的按层切分的流水线并行(Pipeline Parallelism, PP). 由于它打破了单卡放完整模型的约束, 官方示例将其归类为模型并行(Model Parallelism)的大范畴.

2. 在现代 LLM 训练中的实际地位:

   在千亿参数大模型(如 LLaMA、DeepSeek)的真实训练中, 通常会同时组合三种并行方式(3D 混合并行):

   * 张量并行 (TP): 在单机 8 卡内部切分单个 Transformer Layer 的 Attention/MLP 矩阵(NVLink 速度够快);

   * 流水线并行 (PP): 将百层 Transformer 按层切分到不同的机器节点上;

   * 数据并行 / FSDP (DP): 在剩余的卡维度上对训练数据 Batch 进行切分, 使用类似 DDP / FSDP 的方式同步梯度.
